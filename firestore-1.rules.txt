rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function owner(uid) { return signedIn() && request.auth.uid == uid; }
    function room(roomId) { return get(/databases/$(database)/documents/rooms/$(roomId)); }
    function roomExists(roomId) { return exists(/databases/$(database)/documents/rooms/$(roomId)); }
    function isRoomOwner(roomId) {
      return signedIn() && roomExists(roomId) && room(roomId).data.ownerUid == request.auth.uid;
    }

    match /users/{uid} {
      allow read: if owner(uid);
      allow create: if owner(uid);
      allow update: if owner(uid)
        && request.resource.data.plan == resource.data.plan
        && request.resource.data.subscriptionStatus == resource.data.subscriptionStatus
        && request.resource.data.subscriptionExpiresAt == resource.data.subscriptionExpiresAt;
      allow delete: if false;
    }

    match /payments/{id} {
      allow read, write: if false;
    }

    match /rooms/{roomId} {
      allow read: if true;
      allow create: if signedIn() && request.resource.data.ownerUid == request.auth.uid
        && request.resource.data.retentionDays is int
        && (request.resource.data.retentionDays == 30
            || (request.resource.data.retentionDays == 90
                && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.plan == "pro"));
      allow update, delete: if signedIn() && resource.data.ownerUid == request.auth.uid
        && resource.data.status == "live"
        && request.time < resource.data.createdAt + duration.value(3, "d");

      // Room owners can block a participant's room-local anonymous ID.
      match /blocked/{participantId} {
        allow read: if true;
        allow write: if isRoomOwner(roomId);
      }
    }

    // Anonymous rooms have no server-verified participant identity — anyone
    // holding the room link can post. These rules enforce shape and content
    // limits, not "who" is writing; see /security for that tradeoff.
    match /messages/{messageId} {
      allow read: if roomExists(resource.data.roomId)
        && room(resource.data.roomId).data.status == "live"
        && request.time < room(resource.data.roomId).data.createdAt + duration.value(3, "d");

      allow create: if roomExists(request.resource.data.roomId)
        && room(request.resource.data.roomId).data.status == "live"
        && request.time < room(request.resource.data.roomId).data.createdAt + duration.value(3, "d")
        && request.resource.data.content is string
        && request.resource.data.content.size() > 0
        && request.resource.data.content.size() <= 2000
        && request.resource.data.participantId is string
        && request.resource.data.participantId.size() > 0
        && request.resource.data.participantId.size() <= 24
        && request.resource.data.edited == false;

      allow update: if resource.data.participantId == request.resource.data.participantId
        && request.resource.data.roomId == resource.data.roomId
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.quotedMessageId == resource.data.quotedMessageId
        && request.resource.data.content is string
        && request.resource.data.content.size() > 0
        && request.resource.data.content.size() <= 2000;

      allow delete: if true;
    }

    // Message reports: anyone can file one, nobody can read/alter them from the client.
    match /reports/{reportId} {
      allow create: if request.resource.data.reason is string
        && request.resource.data.reason.size() > 0
        && request.resource.data.reason.size() <= 200;
      allow read, update, delete: if false;
    }

    // Ephemeral "typing…" presence, auto-expired client-side.
    match /typing/{roomId}/users/{participantId} {
      allow read: if true;
      allow write: if request.resource.data.participantId == participantId;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
